async function loadBeats(){ beatStatus.style.display = "block"; beatStatus.className = "loading"; beatStatus.textContent = "Loading beats..."; try{ const { data: profiles, error: profileError } = await supabaseClient .from("artist_profiles") .select("*"); if(profileError){ throw profileError; } const targetName = String(producerName) .trim() .toLowerCase(); const producerProfile = (profiles || []).find( profile => { const names = [ profile.performing_name, profile.artist_name, profile.stage_name, profile.name ]; return names.some( name => name && String(name) .trim() .toLowerCase() === targetName ); } ); if(!producerProfile){ document .getElementById( "statUploaded" ) .textContent = "0"; renderBeats([]); loadCredits([]); return; } const { data, error } = await supabaseClient .from("songs") .select("*") .eq( "artist_id", producerProfile.id ) .eq( "status", "approved" ) .order( "created_at", { ascending:false } ); if(error){ throw error; } const beats = (data || []) .filter( row => isBeat(row) ); document .getElementById( "statUploaded" ) .textContent = beats.length; renderBeats( beats ); setupBeatSearch( beats ); loadCredits( data || [] ); }catch(error){ console.error( "Beat Store loading error:", error ); beatGrid.innerHTML = ""; beatStatus.style.display = "block"; beatStatus.className = "error"; beatStatus.textContent = "Unable to load beats right now."; } }
