async function uploadProfilePicture(event){ const status = document.getElementById("profileUploadStatus"); try{ const file = event && event.target && event.target.files ? event.target.files[0] : null; if(!file){ return; } status.textContent = "Step 1/4: Checking login..."; const sessionResult = await supabaseClient.auth.getSession(); const session = sessionResult.data.session; if(!session){ status.textContent = "❌ Please log in again."; alert("Please log in again."); return; } status.textContent = "Step 2/4: Getting upload permission..."; const signatureResponse = await fetch( API_BASE + "/api/cloudinary/signature?folder=pasong-profiles", { method:"GET", mode:"cors", cache:"no-store" } ); if(!signatureResponse.ok){ throw new Error( "Signature request failed: HTTP " + signatureResponse.status ); } const signatureData = await signatureResponse.json(); if( !signatureData.success || !signatureData.signature ){ throw new Error( "Server did not return a valid Cloudinary signature." ); } status.textContent = "Step 3/4: Uploading picture..."; const form = new FormData(); form.append( "file", file ); form.append( "api_key", signatureData.api_key ); form.append( "timestamp", String(signatureData.timestamp) ); form.append( "signature", signatureData.signature ); form.append( "folder", signatureData.folder ); form.append( "public_id", signatureData.public_id ); const uploadUrl = signatureData.upload_url || signatureData.uploadUrl; if(!uploadUrl){ throw new Error( "Cloudinary upload URL is missing." ); } let cloudinaryResponse; try{ cloudinaryResponse = await fetch( uploadUrl, { method:"POST", mode:"cors", body:form } ); }catch(networkError){ throw new Error( "Cloudinary connection failed. " + networkError.message ); } const cloudinaryText = await cloudinaryResponse.text(); let cloudinaryData; try{ cloudinaryData = JSON.parse(cloudinaryText); }catch(error){ throw new Error( "Cloudinary returned an invalid response." ); } if( !cloudinaryResponse.ok || !cloudinaryData.secure_url ){ throw new Error( cloudinaryData.error && cloudinaryData.error.message ? cloudinaryData.error.message : "Cloudinary rejected the picture." ); } const imageUrl = cloudinaryData.secure_url; status.textContent = "Step 4/4: Saving profile..."; let saveResponse; try{ saveResponse = await fetch( API_BASE + "/api/artist-profile/save", { method:"POST", mode:"cors", headers:{ "Content-Type":"application/json", "Authorization": "Bearer " + session.access_token }, body:JSON.stringify({ profile_image_url:imageUrl }) } ); }catch(networkError){ throw new Error( "Picture uploaded, but PASONG could not connect to the profile-saving server. " + networkError.message ); } const saveText = await saveResponse.text(); let saveData; try{ saveData = JSON.parse(saveText); }catch(error){ throw new Error( "Profile server returned an invalid response: " + saveText.substring(0,150) ); } if(!saveResponse.ok){ throw new Error( saveData.error || "Profile could not be saved." ); } document.getElementById( "profilePreview" ).innerHTML = ''; status.textContent = "✓ Profile picture uploaded successfully."; alert( "Profile picture uploaded successfully." ); document.getElementById( "profilePictureInput" ).value = ""; }catch(error){ console.error( "PROFILE UPLOAD ERROR:", error ); status.textContent = "❌ " + error.message; alert( "Profile picture upload failed:\n\n" + error.message ); } }
