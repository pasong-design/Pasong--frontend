async function uploadProfilePicture() { const fileInput = document.getElementById("profilePictureInput"); if (!fileInput || !fileInput.files || !fileInput.files[0]) { alert("Please choose a profile picture first."); return; } const file = fileInput.files[0]; if (!file.type.startsWith("image/")) { alert("Please select an image file."); return; } if (file.size > 5 * 1024 * 1024) { alert("Profile picture must be 5MB or smaller."); return; } try { const { data: { session }, error: sessionError } = await supabase.auth.getSession(); if (sessionError || !session) { throw new Error("Please log in again before uploading your profile picture."); } /* ----------------------------------------- STEP 1: GET CLOUDINARY SIGNATURE ----------------------------------------- */ const signResponse = await fetch( API_BASE + "/api/cloudinary/signature?folder=pasong-profiles" ); if (!signResponse.ok) { throw new Error( "Could not get Cloudinary upload signature." ); } const uploadData = await signResponse.json(); if (!uploadData.success) { throw new Error( uploadData.error || "Cloudinary signature failed." ); } /* ----------------------------------------- STEP 2: UPLOAD IMAGE TO CLOUDINARY ----------------------------------------- */ const cloudinaryForm = new FormData(); cloudinaryForm.append("file", file); cloudinaryForm.append( "api_key", uploadData.api_key ); cloudinaryForm.append( "timestamp", String(uploadData.timestamp) ); cloudinaryForm.append( "signature", uploadData.signature ); cloudinaryForm.append( "folder", uploadData.folder ); cloudinaryForm.append( "public_id", uploadData.public_id ); const cloudinaryResponse = await fetch( uploadData.upload_url, { method: "POST", body: cloudinaryForm } ); const cloudinaryResult = await cloudinaryResponse.json(); if ( !cloudinaryResponse.ok || !cloudinaryResult.secure_url ) { throw new Error( cloudinaryResult.error?.message || "Image upload to Cloudinary failed." ); } const imageUrl = cloudinaryResult.secure_url; /* ----------------------------------------- STEP 3: SAVE URL THROUGH PASONG API ----------------------------------------- */ const saveResponse = await fetch( API_BASE + "/api/artist-profile/save", { method: "POST", headers: { "Content-Type": "application/json", "Authorization": "Bearer " + session.access_token }, body: JSON.stringify({ profile_image_url: imageUrl }) } ); const saveText = await saveResponse.text(); let saveResult; try { saveResult = JSON.parse(saveText); } catch (e) { throw new Error( "PASONG API returned an invalid response." ); } if ( !saveResponse.ok || !saveResult.success ) { throw new Error( saveResult.error || "Profile image could not be saved." ); } /* ----------------------------------------- STEP 4: SHOW IMAGE ----------------------------------------- */ const profileImages = document.querySelectorAll( "#profilePreview, .profile-preview, .profile-image" ); profileImages.forEach(function (img) { if (img.tagName === "IMG") { img.src = imageUrl; } }); const profilePicture = document.getElementById( "profilePicture" ); if ( profilePicture && profilePicture.tagName === "IMG" ) { profilePicture.src = imageUrl; } fileInput.value = ""; alert( "Profile picture uploaded and saved successfully!" ); } catch (error) { console.error( "Profile picture upload error:", error ); alert( "Profile picture upload failed: " + error.message ); } }
