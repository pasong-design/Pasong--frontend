const PASONG_API = "https://pasong-api.onrender.com"; let pasongStreamSessionId = null; let pasongHeartbeatTimer = null; let pasongStartingSession = false; async function getStreamAuthHeaders() { const { data } = await db.auth.getSession(); const token = data?.session?.access_token; return { "Content-Type": "application/json", ...(token ? { Authorization: `Bearer ${token}` } : {}) }; } async function startPasongStreamSession(song) { if (!song || song._pasong_type === "beat") return; if (pasongStreamSessionId || pasongStartingSession) return; pasongStartingSession = true; try { const headers = await getStreamAuthHeaders(); const response = await fetch( `${PASONG_API}/api/streaming/session/start`, { method: "POST", headers, body: JSON.stringify({ song_id: song.id, source_type: "song" }) } ); const result = await response.json(); if (!response.ok || !result.success || !result.session?.id) { throw new Error(result.message || "Could not start streaming session"); } pasongStreamSessionId = result.session.id; clearInterval(pasongHeartbeatTimer); pasongHeartbeatTimer = setInterval(() => { sendPasongHeartbeat(true); }, 15000); sendPasongHeartbeat(true); } catch (error) { console.warn("PASONG streaming session:", error.message); } finally { pasongStartingSession = false; } } async function sendPasongHeartbeat(isPlaying) { if (!pasongStreamSessionId) return; try { const headers = await getStreamAuthHeaders(); await fetch( `${PASONG_API}/api/streaming/session/heartbeat`, { method: "POST", headers, body: JSON.stringify({ session_id: pasongStreamSessionId, media_position_seconds: Number(audio.currentTime || 0), is_playing: Boolean(isPlaying && !audio.paused) }) } ); } catch (error) { console.warn("PASONG heartbeat:", error.message); } } async function stopPasongStreamSession() { clearInterval(pasongHeartbeatTimer); pasongHeartbeatTimer = null; if (pasongStreamSessionId) { await sendPasongHeartbeat(false); } pasongStreamSessionId = null; pasongStartingSession = false; }
